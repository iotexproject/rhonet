    function testWithdrawBeforeRegisterPreservesPriorDeposits() public {
        _fund(SPONSOR, POOL);
        _solveAndSettle(Fixture.TOTAL);
        (address who, uint256 steps, bytes32[] memory proof) = _fixture(0);
        vm.prank(who);
        vault.withdraw();
